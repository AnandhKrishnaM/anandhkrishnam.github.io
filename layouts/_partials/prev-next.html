{{- /* Previous / next. Lectures follow the series order (weight); everything else is chronological within its section. */ -}} {{- $prev := false }}{{ $next := false }} {{- if .Section }} {{- $list := cond (eq .Section "lectures") .CurrentSection.RegularPages.ByWeight .CurrentSection.RegularPages.ByDate }} {{- $i := -1 }}{{ range $k, $p := $list }}{{ if eq $p $ }}{{ $i = $k }}{{ end }}{{ end }} {{- if gt $i 0 }}{{ $prev = index $list (sub $i 1) }}{{ end }} {{- if and (ge $i 0) (lt (add $i 1) (len $list)) }}{{ $next = index $list (add $i 1) }}{{ end }} {{- end }} {{- if or $prev $next }}
{{- with $prev }}← Previous{{ .Title }}{{ else }}{{ end }} {{- with $next }}Next →{{ .Title }}{{ end }}
{{- end }}
